Q14: Write a function that takes an array of numbers and returns the sum of all elements.

function sumarray(arr) { let sum = 0; for (let i = 0; i < arr.length; i++) { sum = sum + arr[i]; } return sum; } console.log(sumarray([10, 20, 30, 40, 50]));
